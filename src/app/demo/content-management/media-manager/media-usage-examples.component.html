Code Examples
Basic Usage:
// Inject the service
constructor(private mediaManagerDialog: MediaManagerDialogService) {}

// Select multiple files
this.mediaManagerDialog.selectFiles().subscribe(files => {
  console.log('Selected files:', files);
});

// Select single file
this.mediaManagerDialog.selectSingleFile().subscribe(file => {
  console.log('Selected file:', file);
});
Advanced Usage:
// Custom configuration
this.mediaManagerDialog.openMediaManager({
  title: 'Select Images',
  config: {
    allowedFileTypes: ['jpg', 'png'],
    maxFileSize: 5 * 1024 * 1024,
    multipleSelection: true
  }
}).subscribe(result => {
  if (result.action === 'select') {
    console.log('Files:', result.files);
  }
});
